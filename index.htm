  <script>

    // CORE SYSTEM DATA STRUCTURES

    let runnersDB = JSON.parse(localStorage.getItem('ocr_runners_db')) || [];

    let leaderboard = JSON.parse(localStorage.getItem('ocr_leaderboard_db')) || [];

    let tvWindow = null;

    // CATEGORY DETERMINATION RULES

    function determineCategory(gender, birthYear) {

      const currentYear = new Date().getFullYear();

      const age = currentYear - birthYear;

 

 

      if (age < 16) {

        return gender === 'K' ? 'Dzieci K' : 'Dzieci M';

      } else if (age >= 16 && age <= 29) {

        return gender === 'K' ? 'K20' : 'M20';

      } else if (age >= 30 && age <= 39) {

        return gender === 'K' ? 'K30' : 'M30';

      } else if (age >= 40 && age <= 49) {

        return gender === 'K' ? 'K40' : 'M40';

      } else {

        return gender === 'K' ? 'K Masters (50+)' : 'M Masters (50+)';

      }

    }

 

 

    // AUDIO GENERATOR (3-2-1 BEEP)

    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    function playBeep(freq, duration, type = 'sine') {

      if (audioCtx.state === 'suspended') audioCtx.resume();

      const osc = audioCtx.createOscillator();

      const gain = audioCtx.createGain();

      osc.type = type;

      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

      gain.gain.setValueAtTime(0.3, audioCtx.currentTime);

      gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration);

      osc.connect(gain);

      gain.connect(audioCtx.destination);

      osc.start();

      osc.stop(audioCtx.currentTime + duration);

    }

 

 

    // NAVIGATION TABS SWITCHING

    function switchTab(tabId) {

      document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));

      document.querySelectorAll('.tab-btn').forEach(btn => {

        btn.classList.remove('bg-amber-500', 'text-slate-950');

        btn.classList.add('bg-slate-800', 'text-slate-300');

      });

 

 

      document.getElementById(`view-${tabId}`).classList.remove('hidden');

      const activeBtn = document.getElementById(`tabBtn-${tabId}`);

      activeBtn.classList.remove('bg-slate-800', 'text-slate-300');

      activeBtn.classList.add('bg-amber-500', 'text-slate-950');

    }

 

 

    // RUNNER REGISTRATION & MANAGEMENT

    function registerRunner(e) {

      e.preventDefault();

      const firstName = document.getElementById('regFirstName').value.trim();

      const lastName = document.getElementById('regLastName').value.trim();

      const gender = document.getElementById('regGender').value;

      const birthYear = parseInt(document.getElementById('regBirthYear').value);

 

 

      const nextBib = runnersDB.length > 0 ? Math.max(...runnersDB.map(r => r.bib)) + 1 : 101;

      const category = determineCategory(gender, birthYear);

 

 

      const runner = {

        bib: nextBib,

        firstName,

        lastName,

        gender,

        birthYear,

        category,

        qrCode: `BIB-${nextBib}`

      };

 

 

      runnersDB.push(runner);

      saveRunnersDB();

      renderRunnersTable();

      document.getElementById('runnerForm').reset();

    }

 

 

    function saveRunnersDB() {

      localStorage.setItem('ocr_runners_db', JSON.stringify(runnersDB));

    // CSV IMPORT RUNNERS

    function importRunnersCSV(event) {

      const file = event.target.files[0];

      if (!file) return;

 

 

      const reader = new FileReader();

      reader.onload = function(e) {

        const text = e.target.result;

        const lines = text.split('\n');

        let count = 0;

 

 

        lines.forEach((line, index) => {

          if (index === 0 || !line.trim()) return; // Skip Header or Empty

          const cols = line.split(',').map(c => c.trim().replace(/^["']|["']$/g, ''));

          if (cols.length >= 4) {

            const firstName = cols[0];

            const lastName = cols[1];

            const gender = cols[2].toUpperCase().includes('K') ? 'K' : 'M';

            const birthYear = parseInt(cols[3]) || 1995;

            const nextBib = runnersDB.length > 0 ? Math.max(...runnersDB.map(r => r.bib)) + 1 : 101;

          

            runnersDB.push({

              bib: nextBib,

              firstName,

              lastName,

              gender,

              birthYear,

              category: determineCategory(gender, birthYear),

              qrCode: `BIB-${nextBib}`

            });

            count++;

          }

        });

 

 

        saveRunnersDB();

        renderRunnersTable();

        alert(`Pomyślnie zaimportowano ${count} zawodników!`);

      };

      reader.readAsText(file);

    }

    // TRACK ASSIGNMENT VIA SCANNER / BIB INPUT

    function handleTrackScan(trackIdx, val) {

      val = val.trim().toUpperCase();

      if (!val) return;

 

 

      let bibNum = null;

      if (val.startsWith('BIB-')) {

        bibNum = parseInt(val.replace('BIB-', ''));

      } else if (!isNaN(parseInt(val))) {

        bibNum = parseInt(val);

      }

 

 

      if (bibNum) {

        const runner = runnersDB.find(r => r.bib === bibNum);

        if (runner) {

          tracks[trackIdx].runner = runner;

          document.getElementById(`displayName${trackIdx + 1}`).innerText = `#${runner.bib} ${runner.firstName} ${runner.lastName}`;

          document.getElementById(`displayCat1` ? `displayCat${trackIdx + 1}` : '').innerText = `Kat: ${runner.category}`;

          broadcastState();

        }

      }

    }

    // START RACE SEQUENCE

    function startRaceSequence() {

      const btn = document.getElementById('startRaceBtn');

      btn.disabled = true;

      btn.classList.add('opacity-50');

 

 

      let count = 3;

      const countInterval = setInterval(() => {

        if (count > 0) {

          playBeep(440, 0.2);

          count--;

        } else {

          clearInterval(countInterval);

          playBeep(880, 0.6, 'square');

          startAllClocks();

          btn.disabled = false;

          btn.classList.remove('opacity-50');

        }

      }, 1000);

    }

 

 

    function startAllClocks() {

      const now = performance.now();

      tracks.forEach((track, idx) => {

        track.startTime = now;

        track.running = true;

        track.finished = false;

        track.elapsed = 0;

 

 

        document.getElementById(`status${idx + 1}`).innerText = "W Biegu";

        document.getElementById(`status${idx + 1}`).className = "inline-block text-xs uppercase px-2.5 py-1 rounded-full bg-emerald-950 text-emerald-400 font-bold border border-emerald-800";

 

 

        if (track.interval) clearInterval(track.interval);

        track.interval = setInterval(() => {

          if (track.running) {

            track.elapsed = performance.now() - track.startTime;

            document.getElementById(`clock${idx + 1}`).innerText = formatTime(track.elapsed);

          }

        }, 10);

      });

      broadcastState();

    }

 

 

    function stopTrack(idx) {

      const track = tracks[idx];

      if (!track.running || track.finished) return;

 

 

      track.running = false;

      track.finished = true;

      clearInterval(track.interval);

      track.elapsed = performance.now() - track.startTime;

 

 

      document.getElementById(`clock${idx + 1}`).innerText = formatTime(track.elapsed);

      document.getElementById(`status${idx + 1}`).innerText = "Meta";

      document.getElementById(`status${idx + 1}`).className = "inline-block text-xs uppercase px-2.5 py-1 rounded-full bg-cyan-950 text-cyan-400 font-bold border border-cyan-800";

 

 

      // Save Result to Leaderboard

      const runner = track.runner || { bib: '---', firstName: 'Anonim', lastName: `Tor ${idx + 1}`, category: 'OPEN' };

    

      leaderboard.push({

        bib: runner.bib,

        name: `${runner.firstName} ${runner.lastName}`,

        category: runner.category,

        track: idx + 1,

        timeMs: track.elapsed,

        formattedTime: formatTime(track.elapsed)

      });

 

 

      // Sort Leaderboard ascending by time

      leaderboard.sort((a, b) => a.timeMs - b.timeMs);

      localStorage.setItem('ocr_leaderboard_db', JSON.stringify(leaderboard));

 

 

      renderLeaderboard();

      broadcastState();

 

 

      // Clear input & runner for next round

      clearTrack(idx);

    }

 

 

    function formatTime(ms) {

      const totalSec = Math.floor(ms / 1000);

      const min = Math.floor(totalSec / 60);

      const sec = totalSec % 60;

      const hundredths = Math.floor((ms % 1000) / 10);

 

 

      return `${String(min).padStart(2, '0')}:${String(sec).padStart(2, '0')}.${String(hundredths).padStart(2, '0')}`;

    }
